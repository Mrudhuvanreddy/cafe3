import { Component } from '@angular/core'; import { FormsModule, NgForm } from '@angular/forms'; @Component({ selector: 'app-week13-forms', templateUrl: './week13-forms.html' }) export class Week13Component { formData = { fullName: '', email: '', phone: '', itemName: 'Cappuccino', quantity: 1, tableNumber: 1, membershipTier: 'Gold', agreeTerms: false }; onSubmit(form: NgForm): void { if (form.valid) { alert(`Order submitted for ${this.formData.fullName}!`); form.resetForm(); } } }
LAB VIVA QUICK REVISION - WEEK 13: Q1: What are the two types of forms in Angular? A: 1. Template-Driven Forms: Simple forms where validation logic is defined inside the template using directives like [(ngModel)], required, and #form="ngForm". 2. Reactive Forms: More scalable and testable, created programmatically in the component class using FormGroup, FormControl, and FormBuilder. Q2: What is the purpose of #orderForm="ngForm"? A: It creates a template reference variable that exports the NgForm directive instance, allowing access to the form's validity ($valid), dirty state ($dirty), and submission methods. Q3: Explain the Angular form validation states. A: - valid: All validation constraints are satisfied. - invalid: At least one validation rule failed. - pristine: The user has not yet changed the control's value. - dirty: The user has modified the control's value. - touched: The user has focused and blurred out of the input element. - untouched: The user has not yet lost focus from the input element.
